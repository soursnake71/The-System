cmake_minimum_required(VERSION 3.22)
project(synsonics CXX C)
set(CMAKE_CXX_STANDARD 17)
add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(synsonics SHARED main.cpp)
target_include_directories(synsonics PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../core)
target_compile_options(synsonics PRIVATE -O3 -Wall)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -u ANativeActivity_onCreate")
target_link_libraries(synsonics native_app_glue android log aaudio)
