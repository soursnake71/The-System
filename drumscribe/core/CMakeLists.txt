# Drum Scribe core engine (C++17). Song in -> drums + bass separated -> every hit / note -> groove maths -> MIDI.
#   cmake -S drumscribe/core -B build-ds -DCMAKE_BUILD_TYPE=Release && cmake --build build-ds -j
#   ctest --test-dir build-ds
# -DDS_WITH_SEPARATION=OFF builds everything except the neural separation (tests don't need it).
cmake_minimum_required(VERSION 3.22)
project(DrumScribeCore VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(DS_WITH_SEPARATION "Demucs (demucs.cpp) + DrumSep (ONNX Runtime) separation" ON)

include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(json)

file(GLOB DS_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(FILTER DS_SOURCES EXCLUDE REGEX "separate\\.cpp$|server_main\\.cpp$|cli_main\\.cpp$")
add_library(dscore STATIC ${DS_SOURCES})
target_include_directories(dscore PUBLIC include)
target_link_libraries(dscore PUBLIC nlohmann_json::nlohmann_json)
find_package(Threads REQUIRED)
target_link_libraries(dscore PUBLIC Threads::Threads)

if(DS_WITH_SEPARATION)
  include(cmake/separation.cmake)  # defines dsseparate (src/separate.cpp + demucs.cpp + onnxruntime)
endif()

# Android (drumscribe/android builds this folder as part of libdrumscribe.so): only the libraries
if(ANDROID)
  return()
endif()

# the engine programs (server + command line), when their sources exist
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3 GIT_SHALLOW TRUE)
set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE INTERNAL "")
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE INTERNAL "")
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE INTERNAL "")
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(httplib)
# the browser page, compiled into the server
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/web/index.html DS_WEB_PAGE)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/gen/web_page.inc "static const char* WEB_PAGE = R\"DSHTML(${DS_WEB_PAGE})DSHTML\";\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS web/index.html)
foreach(prog server cli)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/${prog}_main.cpp)
    add_executable(drumscribe-${prog} src/${prog}_main.cpp)
    target_link_libraries(drumscribe-${prog} PRIVATE dscore httplib::httplib)
    target_include_directories(drumscribe-${prog} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/gen src)
    # an upload sent without a content type is parsed as a form: let those be as big as any audio file
    target_compile_definitions(drumscribe-${prog} PRIVATE CPPHTTPLIB_FORM_URL_ENCODED_PAYLOAD_MAX_LENGTH=1073741824)
    if(WIN32)
      target_link_libraries(drumscribe-${prog} PRIVATE ws2_32)
    endif()
    if(TARGET dsseparate)
      target_link_libraries(drumscribe-${prog} PRIVATE dsseparate)
      target_compile_definitions(drumscribe-${prog} PRIVATE DS_HAVE_SEPARATION=1)
      if(COMMAND ds_copy_onnxruntime)
        ds_copy_onnxruntime(drumscribe-${prog})
      endif()
    endif()
  endif()
endforeach()

enable_testing()
file(GLOB DS_TESTS CONFIGURE_DEPENDS tests/test_*.cpp)
foreach(t ${DS_TESTS})
  get_filename_component(n ${t} NAME_WE)
  add_executable(${n} ${t})
  target_link_libraries(${n} PRIVATE dscore)
  target_include_directories(${n} PRIVATE tests)
  add_test(NAME ${n} COMMAND ${n})
endforeach()
